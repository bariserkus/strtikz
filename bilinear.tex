\newcommand{\bilinear}[1][]{
\pgfkeys{/bilinear, default, #1}
\tikzmath{
real \startx, \starty;
real \dy, \fy, \dm, \fm, \axsp;
real \axislinet, \bilinet, \borderlinethk;
integer \mode;
%Conversion all units to the unit pt
\dy = \dy;
\fy = \fy;
\dm = \dm;
\fm = \fm;
\axsp = \axsp;
\axislinet =\axislinet;
\bilinet = \bilinet;
\arrlen = \arrlen;
\arrwid = \arrwid;
\borderlinethk=\borderlinethk;
%End conversion, All values are now in pt units
\xmax = \dm + \axsp;
\ymax = \fm + \axsp;
\pyd = \dm - \dy;   %Post yield displacement
\pyf = \fm - \fy;   %Post yield force
\lowlevel = min(-\axsp,\fm - 2*\fy - \axsp)-0.5cm;
if \showaxes==0 then {\xlab={};\ylab={};} else {};
if \showstiff==0 then {\kpt={};\kst={};} else {};
%% Mode4 minx
\modefourx=\dm- 2*\dy -\pyd -\axsp;
\modefoury=\fm -2*\fy -\pyf -\axsp;
%% Borders
if \mode==1 then {\bordermaxx=\xmax; \borderminx=-\axsp;} else {};
if \mode==2 then {\bordermaxx=\xmax; \borderminx=-\axsp;} else {};
if \mode==3 then {\bordermaxx=\xmax; \borderminx=-\axsp;} else {};
if \mode==4 then {\bordermaxx=\xmax; \borderminx=\modefourx;} else {};
if \mode==5 then {\bordermaxx=\xmax; \borderminx=-\xmax;} else {};
if \mode==6 then {\bordermaxx=\xmax; \borderminx=-\xmax;} else {};
if \mode==7 then {\bordermaxx=\xmax; \borderminx=-\xmax;} else {};
if \mode==8 then {\bordermaxx=\xmax; \borderminx=-\xmax;} else {};
\borderslope=(\fm-\fy)/(\dm-\dy);
\qy=\fy-\borderslope*\dy;
\bordertopmaxy=\qy+\borderslope*\bordermaxx;
\bordertopminy=\qy+\borderslope*\borderminx;
\borderbotmaxy=-\qy+\borderslope*\bordermaxx;
\borderbotminy=-\qy+\borderslope*\borderminx;
%% Region Text
\regiontextx=(\dm-\dy)/3+\dy;
\regiontexty=(\fy)/3;
}

\begin{scope}[x=1pt, y=1pt, xshift = \startx, yshift=\starty, rotate=0];    % Draw everything in pt units

\ifthenelse{\showshade=1}{

\fill[opacity=\shadeopacity, \shadecolor] (\borderminx, \bordertopminy) -- (\bordermaxx, \bordertopmaxy) --
(\bordermaxx, \borderbotmaxy) -- (\borderminx, \borderbotminy) -- cycle;

\node[\shadecolor, text width=1.5cm, align=center]  at (\regiontextx,\regiontexty) {Interior\\Region};

}{}

\ifthenelse{\showlinearrows=1}{
\tikzset{linearrow/.style = {postaction={on each segment={mid arrow=black}}}};
}{\tikzset{linearrow/.style = {}};}

\ifthenelse{\showarrows=1}{
\tikzset{arrowbe/.style = {arrows={->[length=\arrlen, width=\arrwid]}}};
\tikzset{arrowben/.style = {arrows={<[length=\arrlen, width=\arrwid]->[length=\arrlen, width=\arrwid]}}};
}{\tikzset{arrowbe/.style = {}}; \tikzset{arrowben/.style = {}};}

\ifthenelse{\myp=1}{
\draw [very thin, dashed] (\dy, 0)--node[at start, below=-1]{\dyt}(\dy, \fy);
\draw [very thin, dashed] (0, \fy)--node[at start, inner sep=2pt, outer sep=0pt, anchor=east, yshift=2pt]
{\fyt}(\dy, \fy);
}{}

\ifthenelse{\mmp=1}{
\draw [very thin, dashed] (\dm, 0)--
node[at start, inner sep=2pt, outer sep=0pt, anchor=north]{\dmt}(\dm, \fm);
\draw [very thin, dashed] (0, \fm)--
node[at start, inner sep=2pt, outer sep=0pt, anchor=east]{\fmt}(\dm, \fm);
}{}

\ifthenelse{\mode=1}{
\draw [line width=\axislinet, axesarr] (-\axsp,0) -- node[at end, anchor=south east]{\xlab}(\xmax,0);
\draw [line width=\axislinet, axesarr] (0,-\axsp) -- node[at end, anchor=north east]{\ylab}(0,\ymax);
\path [line width=\bilinet, draw=\linecolor, linearrow]
(0,0)
-- node[right, midway, behind path, inner sep=2pt, outer sep=0.5pt, anchor=west]{\kpt}(\dy,\fy)
-- node[below, midway]{\kst}(\dm,\fm);

}{}

\ifthenelse{\mode=2}{
\draw [line width = \axislinet, axesarr] (-\axsp,0) -- node[at end, anchor=south east]{\xlab}(\xmax,0);
\draw [line width = \axislinet, axesarr] (0,-\axsp) -- node[at end, anchor=north east]{\ylab}(0,\ymax);
\path [line width = \bilinet, draw=black, linearrow]
(0,0)
-- node[right, midway, behind path, inner sep=2pt, outer sep=0.5pt, anchor=west]{\kpt}(\dy,\fy)
-- node[below, midway]{\kst}(\dm,\fm)
-- node[left, midway]{\kpt}++(-\dy, -\fy);
}{}



\ifthenelse{\mode=3}{
\draw [line width = \axislinet, axesarr] (-\axsp,0) -- node[at end, anchor=south east]{\xlab}(\xmax,0);
\draw [line width = \axislinet, twoaxesarr] (0,\lowlevel) -- node[at end, anchor=north east]{\ylab}(0,\ymax);
\path [line width = \bilinet, draw=black, linearrow]
(0,0)
-- node[right, midway, behind path, inner sep=2pt, outer sep=0.5pt, anchor=west]{\kpt}(\dy,\fy)
-- node[below, midway]{\kst}(\dm,\fm)
-- node[left, midway]{\kpt}++(-2*\dy, -2*\fy);
}{}

\ifthenelse{\mode=4}{
\draw [line width = \axislinet, twoaxesarr] (\modefourx,0) --
node[at end, anchor=north]{\xlab}(\xmax,0);
\draw [line width = \axislinet, twoaxesarr] (0,\modefoury) --
node[at end, anchor=east]{\ylab}(0,\ymax);

\path [line width = \bilinet, draw=black, linearrow] 
(0,0)
-- node[right, midway, behind path, inner sep=2pt, outer sep=0.5pt, anchor=west]{\kpt}(\dy,\fy)
-- node[below, midway]{\kst}(\dm,\fm)
-- node[left, midway]{\kpt}++(-2*\dy, -2*\fy)
-- node[above, midway]{\kst}++(-\pyd, -\pyf);
}{}


\ifthenelse{\mode=5}{
\draw [line width = \axislinet, twoaxesarr] (-\xmax,0) -- node[at end, anchor=south east]{\xlab}(\xmax,0);
\draw [line width = \axislinet, twoaxesarr] (0,-\ymax) -- node[at end, anchor=north east]{\ylab}(0,\ymax);
\path [line width = \bilinet, draw=black, linearrow] 
(0,0)
-- node[right, midway, behind path, inner sep=2pt, outer sep=0.5pt, anchor=west]{\kpt}(\dy,\fy)
-- node[below, midway]{\kst}(\dm,\fm)
-- node[left, midway]{\kpt}++(-2*\dy, -2*\fy)
-- node[above, midway]{\kst}++(-2*\pyd, -2*\pyf);
}{}

\ifthenelse{\mode=6}{
\draw [line width = \axislinet, twoaxesarr] (-\xmax,0) -- node[at end, anchor=south east]{\xlab}(\xmax,0);
\draw [line width = \axislinet, twoaxesarr] (0,-\ymax) -- node[at end, anchor=north east]{\ylab}(0,\ymax);
\path [line width = \bilinet, draw=black, linearrow] 
(0,0)
-- node[right, midway, behind path, inner sep=2pt, outer sep=0.5pt, anchor=west]{\kpt}(\dy,\fy)
-- node[below, midway]{\kst}(\dm,\fm)
-- node[left, midway]{\kpt}++(-2*\dy, -2*\fy)
-- node[above, midway]{\kst}++(-2*\pyd, -2*\pyf)
-- node[right, midway]{\kpt}++(\dy, \fy);
}{}

\ifthenelse{\mode=7}{
\draw [line width = \axislinet, twoaxesarr] (-\xmax,0) -- node[at end, anchor=south east]{\xlab}(\xmax,0);
\draw [line width = \axislinet, twoaxesarr] (0,-\ymax) -- node[at end, anchor=north east]{\ylab}(0,\ymax);
\path [line width = \bilinet, draw=black, linearrow] 
(0,0)
-- node[right, midway, behind path, inner sep=2pt, outer sep=0.5pt, anchor=west]{\kpt}(\dy,\fy)
-- node[below, midway]{\kst}(\dm,\fm)
-- node[left, midway]{\kpt}++(-2*\dy, -2*\fy)
-- node[above, midway]{\kst}++(-2*\pyd, -2*\pyf)
-- node[right, midway]{\kpt}++(2*\dy, 2*\fy);
}{}

\ifthenelse{\mode=8}{
\draw [line width = \axislinet, twoaxesarr] (-\xmax,0) -- node[at end, anchor=south east]{\xlab}(\xmax,0);
\draw [line width = \axislinet, twoaxesarr] (0,-\ymax) -- node[at end, anchor=north east]{\ylab}(0,\ymax);
\path [line width = \bilinet, draw=\linecolor, linearrow] 
(0,0)
-- node[right, midway, behind path, inner sep=2pt, outer sep=0.5pt, anchor=west]{\kpt}(\dy,\fy)
-- node[below, midway]{\kst}(\dm,\fm)
-- node[left, midway]{\kpt}++(-2*\dy, -2*\fy)
-- node[above, midway]{\kst}++(-2*\pyd, -2*\pyf)
-- node[right, midway]{\kpt}++(2*\dy, 2*\fy)
-- node[below, midway]{\kst}++(\pyd, \pyf) ;
}{}

\ifthenelse{\showtopborder=1}{
\draw[line width=\borderlinethk, \borderlinecolor, \borderlinedash, opacity=\borderlineopacity]
(\borderminx, \bordertopminy) --
node[above, inner sep=1pt, pos=0.925, sloped, opacity=1, fill=white]{Top Border} (\bordermaxx, \bordertopmaxy);
}{}

\ifthenelse{\showbotborder=1}{
\draw[line width=\borderlinethk, \borderlinecolor, \borderlinedash, opacity=\borderlineopacity]
(\borderminx, \borderbotminy) --
node[below, inner sep=1pt, pos=0.90, sloped, opacity=1, fill=white]{Bottom Border}(\bordermaxx, \borderbotmaxy);
}{}

\ifthenelse{\showqy=1}{

\draw[thin](-4pt, \qy) -- (4pt, \qy);
\fill[] (0, \qy) circle [radius=1pt];

\draw[very thin, dashed] (\dy,\fy) --
node[at end, inner sep=1pt, anchor=north east, yshift=-2pt]{\qyt}(0,\qy);

}{}




\end{scope}
}
